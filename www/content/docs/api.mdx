## Singleton

### afterlog.configure()

```typescript
afterlog.configure(config: SingletonConfig<T>): void
```

Configure the global instance. Call once at startup.

### afterlog.createBuilder()

```typescript
afterlog.createBuilder(init?: Init): Builder<T>
```

Create a builder for a request.

### afterlog.finalize()

```typescript
afterlog.finalize(builder: Builder<T>): Promise<void>
```

Finalize builder and emit if sampled.

### afterlog.flush()

```typescript
afterlog.flush(): Promise<void>
```

Flush buffered events.

### afterlog.destroy()

```typescript
afterlog.destroy(): Promise<void>
```

Clean up resources.

### afterlog.isHealthy()

```typescript
afterlog.isHealthy(): Promise<boolean>
```

Check if adapters are working.

### afterlog.getMetrics()

```typescript
afterlog.getMetrics(): AdapterMetrics | null
```

Get adapter metrics.

## Types

### WideEvent

```typescript
interface WideEvent extends HttpWideEvent {
  [key: string]: unknown
}
```

### BaseWideEvent

```typescript
interface BaseWideEvent {
  readonly request_id: UUID
  readonly trace_id: UUID
  readonly timestamp: string
  readonly service?: string
  readonly version?: string
  readonly region?: string
  readonly deployment_id?: string
  readonly environment?: string
}
```

### HttpWideEvent

```typescript
interface HttpWideEvent extends BaseWideEvent {
  http_method?: string
  path?: string
  query?: Record<string, string>
  http_status_code?: number
  request_duration_ms?: number
  request_outcome?: "success" | "error" | "timeout" | "cancelled"
  request_bytes?: number
  response_bytes?: number
  error?: WideEventError
  timings?: Record<string, number>
}
```

### WideEventError

```typescript
interface WideEventError {

  stack?: string | null
  retriable?: boolean

}
```

